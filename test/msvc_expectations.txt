# Known deviations when the tests run on Windows with MSVC
# (.github/workflows/msvc.yml, scripts/check_tests_with_compiler.py).
#
# Format: <category> <test> [opt|noopt|execute]   # reason
# Categories: dredd-fail, golden-diff, compile-fail, run-fail.
# Without a variant, the line applies to all variants of the test.
#
# A listed deviation that no longer happens is reported as XPASS; remove it
# then. A failing run uploads msvc_expectations.suggested.txt, which lists
# every failure of that run.

# The golden files were generated on Linux (LP64). On Windows (LLP64), long
# is 32 bits, so 4294967295 is a long long; size_t is unsigned long long.
golden-diff test/single_file/positive_int_as_minus_one.c noopt
golden-diff test/single_file/positive_int_as_minus_one.cc noopt
golden-diff test/single_file/do_not_mutate_under_alignof.cc
golden-diff test/single_file/do_not_mutate_under_sizeof.c
golden-diff test/single_file/sizeof_template.cc
golden-diff test/single_file/sizeof_template2.cc noopt
golden-diff test/single_file/add_type_aliases.c noopt
golden-diff test/single_file/add_type_aliases.cc noopt
golden-diff test/single_file/initializer_list.cc
golden-diff test/single_file/vector_returns_temporary.cc

# The test input holds its __super code under #ifdef _MSC_VER, because only
# Microsoft extensions have __super. The golden files come from Linux, where
# that code is left out and Dredd mutates nothing; MSVC targets mutate it.
golden-diff test/single_file/super_in_lazy_operand.cc

# The test inputs use variable-length arrays, which MSVC does not support.
compile-fail test/single_file/non_const_sized_array.c
compile-fail test/single_file/non_const_sized_array.cc

# The test inputs call __builtin_frame_address, a GCC built-in that MSVC
# does not have.
compile-fail test/single_file/builtin_frame_address.cc
compile-fail test/single_file/builtin_frame_address_with_argument_rewrite.cc
compile-fail test/execute/constant_function_argument execute

# The test input applies alignof to an expression, a GNU extension; MSVC
# accepts only a type (C2061).
compile-fail test/single_file/do_not_mutate_under_alignof.cc

# The test input has a bool function without a return statement; MSVC
# rejects it by default (C4716), Clang only warns.
compile-fail test/single_file/true_template_arg.cc
